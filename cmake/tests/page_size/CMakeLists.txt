cmake_minimum_required(VERSION 3.18)
project(xiaohei_page_size_test C)

if(NOT DEFINED XIAOHEI_PAGE_SIZE_CMAKE)
    set(XIAOHEI_PAGE_SIZE_CMAKE "${CMAKE_CURRENT_LIST_DIR}/../../xiaohei_page_size.cmake")
endif()
if(NOT EXISTS "${XIAOHEI_PAGE_SIZE_CMAKE}")
    message(FATAL_ERROR "missing helper: ${XIAOHEI_PAGE_SIZE_CMAKE}")
endif()
include("${XIAOHEI_PAGE_SIZE_CMAKE}")

set(XIAOHEI_PAGE_SIZE_PROBE_C "${CMAKE_CURRENT_LIST_DIR}/probe.c")
add_subdirectory(deps)

add_library(LlamaWrapper SHARED "${XIAOHEI_PAGE_SIZE_PROBE_C}")

target_link_options(already_max PRIVATE "-Wl,-z,max-page-size=16384")

add_library(interface_only SHARED "${XIAOHEI_PAGE_SIZE_PROBE_C}")
target_link_options(interface_only INTERFACE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_library(debug_only SHARED "${XIAOHEI_PAGE_SIZE_PROBE_C}")
set_target_properties(debug_only PROPERTIES LINK_FLAGS_DEBUG "-Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384")

    IMPORTED_LOCATION "${CMAKE_CURRENT_BINARY_DIR}/libprebuilt_relro_dummy.so"
)

xiaohei_apply_16kb_page_size_graph(LlamaWrapper)
xiaohei_apply_16kb_page_size(already_max ggml_static prebuilt_relro)
xiaohei_apply_16kb_page_size_tree()

function(_xiaohei_assert_both target)
    get_target_property(_opts "${target}" LINK_OPTIONS)
    if(NOT _opts OR _opts STREQUAL "_opts-NOTFOUND")
        message(FATAL_ERROR "${target}: LINK_OPTIONS empty")
    endif()
    string(FIND "${_opts}" "max-page-size=16384" _i_max)
    string(FIND "${_opts}" "common-page-size=16384" _i_common)
    if(_i_max EQUAL -1 OR _i_common EQUAL -1)
        message(FATAL_ERROR "${target}: missing max/common flags: ${_opts}")
    endif()
    string(REGEX MATCHALL "max-page-size=16384" _max_hits "${_opts}")
    string(REGEX MATCHALL "common-page-size=16384" _common_hits "${_opts}")
    list(LENGTH _max_hits _max_n)
    list(LENGTH _common_hits _common_n)
    if(NOT _max_n EQUAL 1 OR NOT _common_n EQUAL 1)
        message(FATAL_ERROR "${target}: duplicate page-size flags (${_max_n}/${_common_n}): ${_opts}")
    endif()
    file(APPEND "${CMAKE_BINARY_DIR}/observed_link_options.txt" "${target}|ok|${_opts}\n")
endfunction()

function(_xiaohei_assert_no_page_flags target)
    get_target_property(_opts "${target}" LINK_OPTIONS)
    if(_opts AND NOT _opts STREQUAL "_opts-NOTFOUND")
        string(FIND "${_opts}" "max-page-size=16384" _i_max)
        string(FIND "${_opts}" "common-page-size=16384" _i_common)
        if(NOT _i_max EQUAL -1 OR NOT _i_common EQUAL -1)
            message(FATAL_ERROR "${target}: STATIC/IMPORTED must not get page-size flags: ${_opts}")
        endif()
    endif()
    file(APPEND "${CMAKE_BINARY_DIR}/observed_link_options.txt" "${target}|skipped|\n")
endfunction()

file(WRITE "${CMAKE_BINARY_DIR}/observed_link_options.txt" "")
_xiaohei_assert_both(LlamaWrapper)
_xiaohei_assert_both(ggml)
_xiaohei_assert_both(ggml_cpu)
_xiaohei_assert_both(already_max)
_xiaohei_assert_both(interface_only)
_xiaohei_assert_both(debug_only)
_xiaohei_assert_no_page_flags(ggml_static)
_xiaohei_assert_no_page_flags(prebuilt_relro)

xiaohei_page_size_link_flags(_flag_list)
list(LENGTH _flag_list _flag_n)
if(NOT _flag_n EQUAL 2)
    message(FATAL_ERROR "xiaohei_page_size_link_flags must return 2 flags, got ${_flag_n}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/page_size_flags.txt" "${_flag_list}\n")
message(STATUS "xiaohei_page_size_test: LINK_OPTIONS assertions passed")
